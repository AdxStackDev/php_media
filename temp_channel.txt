  <script>
    const channels = <?php echo json_encode($channels); ?>;
    const channelKeys = Object.keys(channels);
    let currentIndex = -1;

    // Store the current HLS instance globally
    let currentHls = null;

    function playChannel(channelUrl, channelName) {
      if (!channelUrl) {
        alert(`${channelName}: Channel link not available.`);
        return;
      }

      const video = document.getElementById('videoPlayerElement');
      
      // Destroy any existing HLS instance to prevent memory leaks and ensure clean switching
      if (currentHls) {
        currentHls.destroy();
        currentHls = null;
      }
      
      // Reset the video element
      video.pause();
      video.removeAttribute('src');
      video.load();
      
      // Create a new HLS instance
      const hls = new Hls();
      currentHls = hls;
      
      hls.on(Hls.Events.ERROR, function (event, data) {
        if (data.fatal) {
          switch(data.type) {
            case Hls.ErrorTypes.NETWORK_ERROR:
              hls.startLoad();
              break;
            case Hls.ErrorTypes.MEDIA_ERROR:
              hls.recoverMediaError();
              break;
            default:
              hls.destroy();
              currentHls = null;
              break;
          }
        }
      });

      hls.loadSource(channelUrl);
      hls.attachMedia(video);

      hls.on(Hls.Events.MANIFEST_PARSED, function() {
        video.play().catch(e => console.error('Error playing video:', e));
        $('#videoSection').removeClass('hidden');
        $('#channelGrid').addClass('hidden');
      });
      
      document.getElementById('currentChannelName').textContent = channelName;
      currentIndex = channelKeys.indexOf(channelName);
      addToRecent(channelName);
    }

    function navigateChannel(direction) {
      if (currentIndex === -1) return;
      currentIndex += direction;
      if (currentIndex < 0) currentIndex = channelKeys.length - 1;
      if (currentIndex >= channelKeys.length) currentIndex = 0;

      const nextChannelName = channelKeys[currentIndex];
      const nextChannelUrl = channels[nextChannelName].url;
      playChannel(nextChannelUrl, nextChannelName);
    }

    // Search functionality
    $('#searchInput').on('input', function() {
      const searchTerm = $(this).val().toLowerCase();
      $('.channel-card').each(function() {
        const channelName = $(this).find('h3').text().toLowerCase();
        $(this).toggle(channelName.includes(searchTerm));
      });
    });

    // Category filtering
    $('.category-btn').click(function() {
      const category = $(this).data('category');
      if (category === 'all') {
        $('.channel-card').show();
      } else {
        $('.channel-card').hide();
        $(`.channel-card[data-category="${category}"]`).show();
      }
    });

    // Event listeners
    document.getElementById('prevChannel').addEventListener('click', () => navigateChannel(-1));
    document.getElementById('nextChannel').addEventListener('click', () => navigateChannel(1));

    // Favorites functionality
    let favorites = JSON.parse(localStorage.getItem('favorites')) || [];
    
    function toggleFavorite(channelName) {
        const index = favorites.indexOf(channelName);
        if (index === -1) {
            favorites.push(channelName);
        } else {
            favorites.splice(index, 1);
        }
        localStorage.setItem('favorites', JSON.stringify(favorites));
        updateFavoritesUI();
    }

    function updateFavoritesUI() {
        $('.channel-card').each(function() {
            const channelName = $(this).find('h3').text();
            const isFavorite = favorites.includes(channelName);
            const favoriteBtn = $(this).find('.favorite-btn');
            
            if (isFavorite) {
                favoriteBtn.html('<i class="fas fa-heart text-red-500"></i>');
            } else {
                favoriteBtn.html('<i class="far fa-heart"></i>');
            }
        });
    }

    // Recent channels functionality
    let recentChannels = JSON.parse(localStorage.getItem('recentChannels')) || [];
    
    function addToRecent(channelName) {
        recentChannels = recentChannels.filter(channel => channel !== channelName);
        recentChannels.unshift(channelName);
        recentChannels = recentChannels.slice(0, 10);
        localStorage.setItem('recentChannels', JSON.stringify(recentChannels));
    }

    // Quality settings
    function setQuality(quality) {
        const video = document.getElementById('videoPlayerElement');
        if (video) {
            video.quality = quality;
        }
    }

    // Theme settings
    function toggleTheme() {
        document.body.classList.toggle('bg-gray-900');
        document.body.classList.toggle('bg-white');
        document.body.classList.toggle('text-white');
        document.body.classList.toggle('text-gray-900');
    }

    // Event listeners for new buttons
    $('#favoritesBtn').click(function() {
        $('.channel-card').hide();
        favorites.forEach(channelName => {
            $(`.channel-card:contains('${channelName}')`).show();
        });
    });

    $('#recentBtn').click(function() {
        $('.channel-card').hide();
        recentChannels.forEach(channelName => {
            $(`.channel-card:contains('${channelName}')`).show();
        });
    });

    $('#qualityBtn').click(function() {
        const quality = prompt('Enter quality (low, medium, high):');
        if (quality) setQuality(quality);
    });

    $('#themeBtn').click(function() {
        toggleTheme();
    });

    // Update channel card template to include favorite button
    $('.channel-card').each(function() {
        $(this).find('.p-4').append(`
            <button class="favorite-btn mt-2 text-gray-400 hover:text-red-500" 
                    onclick="toggleFavorite('${$(this).find('h3').text()}')">
                <i class="far fa-heart"></i>
            </button>
        `);
    });

    // Location-based regional channels
    let userLocation = {
        state: null,
        region: null
    };

    function getLocation() {
        if (navigator.geolocation) {
            navigator.geolocation.getCurrentPosition(
                async function(position) {
                    try {
                        // Use a reverse geocoding service to get state/region
                        const response = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${position.coords.latitude}&lon=${position.coords.longitude}`);
                        const data = await response.json();
                        
                        // Extract state/region from address
                        const address = data.address;
                        userLocation.state = address.state || address.region || address.county;
                        userLocation.region = address.region || address.state || address.county;
                        
                        // Store in localStorage
                        localStorage.setItem('userLocation', JSON.stringify(userLocation));
                        
                        // Show regional channels based on location
                        showRegionalChannels();
                    } catch (error) {
                        console.error('Error getting location:', error);
                        showAllRegionalChannels();
                    }
                },
                function(error) {
                    console.error('Geolocation error:', error);
                    showAllRegionalChannels();
                }
            );
        } else {
            console.log("Geolocation is not supported by this browser.");
            showAllRegionalChannels();
        }
    }

    function showRegionalChannels() {
        if (!userLocation.state) {
            const savedLocation = localStorage.getItem('userLocation');
            if (savedLocation) {
                userLocation = JSON.parse(savedLocation);
            } else {
                showAllRegionalChannels();
                return;
            }
        }

        $('.channel-card').hide();
        $('.channel-card[data-category="regional"]').each(function() {
            const channelName = $(this).find('h3').text();
            // Check if channel name contains the state/region name
            if (channelName.toLowerCase().includes(userLocation.state.toLowerCase()) ||
                channelName.toLowerCase().includes(userLocation.region.toLowerCase())) {
                $(this).show();
            }
        });
    }

    function showAllRegionalChannels() {
        $('.channel-card').hide();
        $('.channel-card[data-category="regional"]').show();
    }

    // Update regional button click handler
    $('.category-btn[data-category="regional"]').click(function() {
        if (!userLocation.state) {
            getLocation();
        } else {
            showRegionalChannels();
        }
    });

    // Add location info to channel cards
    $('.channel-card[data-category="regional"]').each(function() {
        const channelName = $(this).find('h3').text();
        const locationInfo = channelName.match(/(DD|TV)\s+(\w+)/i);
        if (locationInfo) {
            const region = locationInfo[2];
            $(this).attr('data-region', region.toLowerCase());
        }
    });

    // Add location permission request on page load
    $(document).ready(function() {
        const savedLocation = localStorage.getItem('userLocation');
        if (savedLocation) {
            userLocation = JSON.parse(savedLocation);
        } else {
            // Show location permission request
            const locationPermission = confirm('Would you like to see regional channels based on your location?');
            if (locationPermission) {
                getLocation();
            }
        }
    });

    // Add location refresh button
    $('#locationBtn').click(function() {
        getLocation();
    });

    // Update the sidebar to include location refresh
    $('.category-btn[data-category="regional"]').after(`
        <li>
            <button id="locationBtn" class="w-full text-left px-4 py-2 rounded hover:bg-gray-700">
                <i class="fas fa-sync-alt mr-2"></i>Refresh Location
            </button>
        </li>
    `);
  </script>
